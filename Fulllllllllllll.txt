1. What to install
Tool	Version	Why
.NET SDK	10.0.x (latest patch)	Backend, MCP server, evals
Node.js	22 LTS (22.12+) or 24 LTS	Angular
Angular CLI	20.x → npm i -g @angular/cli@20	The frontend code targets Angular 20
PostgreSQL	17.x	Database
pgvector extension	v0.8.1 or newer	Vector search inside Postgres
Ollama	Latest	Runs the LLM locally for free
Ollama models	qwen2.5:7b (~4.7 GB) and mxbai-embed-large (~670 MB)	Chat model with tool calling, plus 1024-dim embeddings
IDE	Visual Studio 2026 (ASP.NET workload) or VS Code + C# Dev Kit + Angular Language Service	Editing
Git	Any recent	Building pgvector on Windows, version control

Hardware: at least 16 GB RAM for qwen2.5:7b. If your machine struggles, use qwen2.5:3b, but its tool calling is noticeably weaker.

Installing pgvector

Windows (pgvector has no installer, so you build it):

Install Visual Studio Build Tools with the “Desktop development with C++” workload.
Open “x64 Native Tools Command Prompt for VS” as Administrator.
Run:
bat
set "PGROOT=C:\Program Files\PostgreSQL\17"
cd %TEMP%
git clone --branch v0.8.1 https://github.com/pgvector/pgvector.git
cd pgvector
nmake /F Makefile.win
nmake /F Makefile.win install

macOS: brew install postgresql@17 pgvector

Ubuntu: sudo apt install postgresql-17 postgresql-17-pgvector (via the PGDG apt repo)

Database setup

Run this once, connected as the postgres superuser in psql or pgAdmin:

sql
CREATE USER opspilot WITH PASSWORD 'opspilot';
CREATE DATABASE opspilot OWNER opspilot;
\c opspilot
CREATE EXTENSION IF NOT EXISTS vector;

using System.Text.Json;
using System.Text.Json.Serialization;

namespace OpsPilot.Core;

public static class Json
{
    public static readonly JsonSerializerOptions Options = new(JsonSerializerDefaults.Web)
    {
        Converters = { new JsonStringEnumConverter() }
    };

    public static string Serialize<T>(T value) => JsonSerializer.Serialize(value, Options);
    public static T? Deserialize<T>(string json) => JsonSerializer.Deserialize<T>(json, Options);

    /// <summary>
    /// LLMs sometimes wrap JSON in prose or ```json fences. Take the outermost {...} and parse it.
    /// </summary>
    public static T ParseLlmJson<T>(string raw)
    {
        var start = raw.IndexOf('{');
        var end = raw.LastIndexOf('}');
        if (start < 0 || end <= start)
            throw new FormatException($"Model did not return JSON. Raw output: {Truncate(raw, 300)}");
        return JsonSerializer.Deserialize<T>(raw[start..(end + 1)], Options)
               ?? throw new FormatException("Model returned empty JSON.");
    }

    public static string Truncate(string? s, int max) =>
        string.IsNullOrEmpty(s) ? "" : s.Length <= max ? s : s[..max] + "…";
}

using Microsoft.EntityFrameworkCore;
using OpsPilot.Core.Domain;

namespace OpsPilot.Core.Data;

public class OpsPilotDbContext(DbContextOptions<OpsPilotDbContext> options) : DbContext(options)
{
    /// <summary>Titan Embed v2 (AWS) and mxbai-embed-large (Ollama) both produce 1024-dim vectors.</summary>
    public const int EmbeddingDimensions = 1024;

    public DbSet<Incident> Incidents => Set<Incident>();
    public DbSet<AgentStep> AgentSteps => Set<AgentStep>();
    public DbSet<DocumentChunk> Chunks => Set<DocumentChunk>();

    protected override void OnModelCreating(ModelBuilder b)
    {
        b.HasPostgresExtension("vector");

        b.Entity<Incident>(e =>
        {
            e.Property(x => x.Status).HasConversion<string>();
            e.Property(x => x.ProposedActionJson).HasColumnType("jsonb");
            e.HasMany(x => x.Steps).WithOne().HasForeignKey(s => s.IncidentId);
            e.HasIndex(x => x.CreatedAt);
        });

        b.Entity<AgentStep>().HasIndex(x => new { x.IncidentId, x.At });

        b.Entity<DocumentChunk>(e =>
        {
            e.Property(x => x.Embedding).HasColumnType($"vector({EmbeddingDimensions})");
            e.HasIndex(x => x.Embedding).HasMethod("hnsw").HasOperators("vector_cosine_ops");
            e.HasIndex(x => x.Source);
        });
    }
}

using Amazon;
using Amazon.BedrockRuntime;
using Microsoft.Extensions.AI;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Options;
using OllamaSharp;

namespace OpsPilot.Core.AI;

/// <summary>
/// Everything downstream depends only on IChatClient / IEmbeddingGenerator (Microsoft.Extensions.AI),
/// so switching Ollama (local) to Bedrock (AWS) is a config change, not a code change.
/// </summary>
public static class AiServiceExtensions
{
    public const string TelemetrySource = "OpsPilot.AI";

    public static IServiceCollection AddAiProviders(this IServiceCollection services)
    {
        services.AddSingleton<IAmazonBedrockRuntime>(sp =>
        {
            var o = sp.GetRequiredService<IOptions<AiOptions>>().Value;
            return new AmazonBedrockRuntimeClient(RegionEndpoint.GetBySystemName(o.AwsRegion));
        });

        services.AddSingleton<IChatClient>(sp =>
        {
            var o = sp.GetRequiredService<IOptions<AiOptions>>().Value;
            IChatClient inner = o.Provider.Equals("Bedrock", StringComparison.OrdinalIgnoreCase)
                ? sp.GetRequiredService<IAmazonBedrockRuntime>().AsIChatClient(o.ChatModel)
                : new OllamaApiClient(new Uri(o.OllamaEndpoint), o.ChatModel);

            // Middleware pipeline: OpenTelemetry (token counts, latency) + logging.
            // Function invocation is added by the Agent Framework's ChatClientAgent itself.
            return inner.AsBuilder()
                .UseOpenTelemetry(sourceName: TelemetrySource)
                .UseLogging()
                .Build(sp);
        });

        services.AddSingleton<IEmbeddingGenerator<string, Embedding<float>>>(sp =>
        {
            var o = sp.GetRequiredService<IOptions<AiOptions>>().Value;
            return o.Provider.Equals("Bedrock", StringComparison.OrdinalIgnoreCase)
                ? sp.GetRequiredService<IAmazonBedrockRuntime>().AsIEmbeddingGenerator(o.EmbeddingModel)
                : new OllamaApiClient(new Uri(o.OllamaEndpoint), o.EmbeddingModel);
        });

        return services;
    }
}
src/OpsPilot.Core/Rag/IngestionService.cs
csharp
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.AI;
using Microsoft.Extensions.Logging;
using Microsoft.SemanticKernel.Text;
using OpsPilot.Core.Data;
using OpsPilot.Core.Domain;
using Pgvector;

namespace OpsPilot.Core.Rag;

/// <summary>
/// Chunk (Semantic Kernel TextChunker) -> embed (IEmbeddingGenerator) -> store (pgvector).
/// </summary>
public sealed class IngestionService(
    IDbContextFactory<OpsPilotDbContext> dbFactory,
    IEmbeddingGenerator<string, Embedding<float>> embedder,
    ILogger<IngestionService> logger)
{
    public async Task<int> IngestAsync(string source, string text, CancellationToken ct = default)
    {
        var lines = TextChunker.SplitMarkDownLines(text, maxTokensPerLine: 100);
        var chunks = TextChunker.SplitMarkdownParagraphs(lines, maxTokensPerParagraph: 350, overlapTokens: 40)
            .Where(c => !string.IsNullOrWhiteSpace(c)).ToList();
        if (chunks.Count == 0) return 0;

        // Prefix each chunk with its source so the embedding captures "which runbook" context.
        var inputs = chunks.Select(c => $"Source: {source}\n{c}").ToList();
        var embeddings = await embedder.GenerateAsync(inputs, cancellationToken: ct);

        await using var db = await dbFactory.CreateDbContextAsync(ct);
        await db.Chunks.Where(c => c.Source == source).ExecuteDeleteAsync(ct); // re-ingest = replace
        for (var i = 0; i < chunks.Count; i++)
        {
            db.Chunks.Add(new DocumentChunk
            {
                Source = source,
                ChunkIndex = i,
                Content = chunks[i],
                Embedding = new Vector(embeddings[i].Vector)
            });
        }
        await db.SaveChangesAsync(ct);
        logger.LogInformation("Ingested {Source}: {Count} chunks", source, chunks.Count);
        return chunks.Count;
    }

    public async Task SeedFromFolderAsync(string folder, CancellationToken ct = default)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        if (await db.Chunks.AnyAsync(ct) || !Directory.Exists(folder)) return;

        foreach (var file in Directory.EnumerateFiles(folder, "*.md", SearchOption.AllDirectories))
        {
            var source = Path.GetRelativePath(folder, file).Replace('\\', '/');
            await IngestAsync(source, await File.ReadAllTextAsync(file, ct), ct);
        }
    }
}
src/OpsPilot.Core/Rag/RetrievalService.cs
csharp
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.AI;
using OpsPilot.Core.Data;
using OpsPilot.Core.Domain;
using Pgvector;
using Pgvector.EntityFrameworkCore;

namespace OpsPilot.Core.Rag;

public sealed class RetrievalService(
    IDbContextFactory<OpsPilotDbContext> dbFactory,
    IEmbeddingGenerator<string, Embedding<float>> embedder)
{
    // Uses its own short-lived DbContext so it is safe to call from agents running in parallel.
    public async Task<IReadOnlyList<RetrievedChunk>> SearchAsync(string query, int topK, CancellationToken ct = default)
    {
        var q = new Vector(await embedder.GenerateVectorAsync(query, cancellationToken: ct));
        await using var db = await dbFactory.CreateDbContextAsync(ct);

        return await db.Chunks
            .OrderBy(c => c.Embedding.CosineDistance(q))
            .Take(topK)
            .Select(c => new RetrievedChunk(c.Source, c.Content, 1 - c.Embedding.CosineDistance(q)))
            .ToListAsync(ct);
    }
}

using System.ComponentModel;
using Microsoft.Extensions.Options;
using OpsPilot.Core.Guardrails;
using OpsPilot.Core.Rag;

namespace OpsPilot.Core.Tools;

/// <summary>In-process tool exposed to the KnowledgeAgent (agentic RAG: the agent decides what to search).</summary>
public sealed class KnowledgeTools(RetrievalService retrieval, IOptions<KnowledgeOptions> options)
{
    [Description("Semantic search over runbooks and past incident reports. Returns the most relevant passages with their source file names.")]
    public async Task<string> SearchRunbooks(
        [Description("Natural-language search query, e.g. 'checkout-api 502 after deployment'")] string query,
        CancellationToken ct = default)
    {
        var hits = await retrieval.SearchAsync(query, options.Value.TopK, ct);
        if (hits.Count == 0) return "No relevant documents found.";

        // Retrieved text is untrusted: wrap it so the model treats it as data, not instructions.
        return string.Join("\n\n", hits.Select(h =>
            $"<document source=\"{h.Source}\" score=\"{h.Score:F2}\">\n{PromptGuard.SanitizeRetrieved(h.Content)}\n</document>"));
    }
}

using Microsoft.Extensions.AI;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Options;
using ModelContextProtocol.Client;

namespace OpsPilot.Core.Tools;

/// <summary>
/// Connects lazily to the OpsPilot.McpTools server and exposes its tools as AIFunctions
/// (McpClientTool inherits from AIFunction, so they plug straight into an agent).
/// </summary>
public sealed class McpToolProvider(IOptions<ToolsOptions> options, ILogger<McpToolProvider> logger) : IAsyncDisposable
{
    private readonly SemaphoreSlim _lock = new(1, 1);
    private McpClient? _client;
    private IList<AITool>? _tools;

    public async Task<IList<AITool>> GetToolsAsync(CancellationToken ct = default)
    {
        if (_tools is not null) return _tools;
        await _lock.WaitAsync(ct);
        try
        {
            if (_tools is not null) return _tools;
            var transport = new HttpClientTransport(new HttpClientTransportOptions
            {
                Endpoint = new Uri(options.Value.McpEndpoint),
                Name = "opspilot-tools"
            });
            _client = await McpClient.CreateAsync(transport, cancellationToken: ct);
            var tools = await _client.ListToolsAsync(cancellationToken: ct);
            logger.LogInformation("Loaded {Count} MCP tools: {Names}", tools.Count, string.Join(", ", tools.Select(t => t.Name)));
            _tools = tools.Cast<AITool>().ToList();
            return _tools;
        }
        finally { _lock.Release(); }
    }

    public async ValueTask DisposeAsync()
    {
        if (_client is not null) await _client.DisposeAsync();
    }
}

using System.Text.RegularExpressions;

namespace OpsPilot.Core.Guardrails;

public static partial class PromptGuard
{
    public const int MaxDescriptionLength = 4000;

    [GeneratedRegex(@"(ignore (all|any|previous|prior) (instructions|rules)|disregard .*instructions|you are now|system prompt|reveal your prompt|act as)",
        RegexOptions.IgnoreCase)]
    private static partial Regex InjectionPattern();

    public static (bool Ok, string? Reason) ValidateUserInput(string title, string description)
    {
        if (string.IsNullOrWhiteSpace(title) || string.IsNullOrWhiteSpace(description))
            return (false, "Title and description are required.");
        if (description.Length > MaxDescriptionLength)
            return (false, $"Description exceeds {MaxDescriptionLength} characters.");
        if (InjectionPattern().IsMatch(title) || InjectionPattern().IsMatch(description))
            return (false, "Input looks like a prompt-injection attempt.");
        return (true, null);
    }

    /// <summary>Neutralise instruction-like text inside retrieved documents (indirect prompt injection).</summary>
    public static string SanitizeRetrieved(string content) =>
        InjectionPattern().Replace(content, "[removed]");
}
src/OpsPilot.Core/Guardrails/ActionPolicy.cs
csharp
using OpsPilot.Core.Domain;

namespace OpsPilot.Core.Guardrails;

/// <summary>Allow-list of what the agents may even propose. Anything else is downgraded to a ticket.</summary>
public static class ActionPolicy
{
    public static readonly HashSet<string> AllowedActions =
        ["rollback_deployment", "restart_service", "scale_service", "create_ticket", "no_action"];

    public static readonly HashSet<string> KnownServices =
        ["checkout-api", "payment-service", "inventory-service", "auth-service"];

    public static (ProposedAction Action, string? Violation) Enforce(ProposedAction proposed)
    {
        if (!AllowedActions.Contains(proposed.ActionType))
            return (Downgrade(proposed, $"Action '{proposed.ActionType}' is not on the allow-list"), "action_not_allowed");

        if (proposed.ActionType != "create_ticket" && proposed.ActionType != "no_action"
            && !KnownServices.Contains(proposed.Service))
            return (Downgrade(proposed, $"Unknown service '{proposed.Service}'"), "unknown_service");

        if (proposed.ActionType == "scale_service"
            && proposed.Parameters?.TryGetValue("replicas", out var r) == true
            && int.TryParse(r, out var n) && n > 10)
            return (proposed with { Parameters = new() { ["replicas"] = "10" } }, "replicas_capped");

        return (proposed, null);
    }

    private static ProposedAction Downgrade(ProposedAction p, string why) => new(
        "create_ticket", p.Service, new() { ["reason"] = why }, $"Policy downgrade: {why}. Original rationale: {p.Rationale}", "low");
}

csharp
using System.Collections.Concurrent;
using System.Threading.Channels;
using OpsPilot.Core.Domain;

namespace OpsPilot.Core.Orchestration;

/// <summary>Fan-out of live agent steps to SSE subscribers (one channel per open browser tab).</summary>
public sealed class IncidentEventBus
{
    private readonly ConcurrentDictionary<Guid, ConcurrentDictionary<Guid, Channel<AgentStepDto>>> _subs = new();

    public (Guid Token, ChannelReader<AgentStepDto> Reader) Subscribe(Guid incidentId)
    {
        var channel = Channel.CreateUnbounded<AgentStepDto>();
        var token = Guid.NewGuid();
        _subs.GetOrAdd(incidentId, _ => new())[token] = channel;
        return (token, channel.Reader);
    }

    public void Unsubscribe(Guid incidentId, Guid token)
    {
        if (_subs.TryGetValue(incidentId, out var map) && map.TryRemove(token, out var ch))
            ch.Writer.TryComplete();
    }

    public void Publish(Guid incidentId, AgentStepDto step)
    {
        if (!_subs.TryGetValue(incidentId, out var map)) return;
        foreach (var ch in map.Values) ch.Writer.TryWrite(step);
    }
}

public sealed class IncidentWorker(IncidentQueue queue, IServiceScopeFactory scopes, ILogger<IncidentWorker> logger)
    : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        await foreach (var id in queue.ReadAllAsync(stoppingToken))
        {
            // Fire-and-track so several incidents can be investigated concurrently.
            _ = Task.Run(async () =>
            {
                try
                {
                    await using var scope = scopes.CreateAsyncScope();
                    await scope.ServiceProvider.GetRequiredService<IncidentOrchestrator>().InvestigateAsync(id, stoppingToken);
                }
                catch (Exception ex) { logger.LogError(ex, "Investigation crashed for {IncidentId}", id); }
            }, stoppingToken);
        }
    }
}

csharp
using System.Diagnostics;
using System.Text;
using Microsoft.Agents.AI;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.AI;
using Microsoft.Extensions.Logging;
using OpsPilot.Core.Agents;
using OpsPilot.Core.Data;
using OpsPilot.Core.Domain;
using OpsPilot.Core.Guardrails;
using OpsPilot.Core.Tools;

namespace OpsPilot.Core.Orchestration;

/// <summary>
/// The multi-agent workflow:
///   Triage  ->  (Knowledge || Diagnostics)  ->  Resolution  ->  policy check  ->  HUMAN APPROVAL  ->  execute
/// Written as explicit C# so every step is observable, testable and resumable.
/// </summary>
public sealed class IncidentOrchestrator(
    OpsPilotDbContext db,
    AgentFactory agents,
    RemediationClient remediation,
    IncidentEventBus bus,
    ILogger<IncidentOrchestrator> logger)
{
    public static readonly ActivitySource Activity = new("OpsPilot.Orchestrator");
    private readonly SemaphoreSlim _dbGate = new(1, 1); // DbContext is not thread-safe; parallel agents share it

    public async Task InvestigateAsync(Guid incidentId, CancellationToken ct)
    {
        using var activity = Activity.StartActivity("investigate_incident");
        activity?.SetTag("incident.id", incidentId);

        var incident = await db.Incidents.FirstAsync(i => i.Id == incidentId, ct);
        try
        {
            // ---------- 1. Triage ----------
            await SetStatusAsync(incident, IncidentStatus.Triaging, ct);
            var triageRaw = await RunAgentAsync(incident, agents.CreateTriageAgent(), IncidentPrompt(incident), ct);
            var triage = Json.ParseLlmJson<TriageResult>(triageRaw);
            incident.Category = triage.Category;
            incident.Severity = triage.Severity;
            incident.Service = triage.Service;
            incident.Summary = triage.Summary;
            await SaveAsync(ct);

            // ---------- 2. Knowledge + Diagnostics in parallel ----------
            await SetStatusAsync(incident, IncidentStatus.Investigating, ct);
            var context = $"{IncidentPrompt(incident)}\n\nTriage: {Json.Serialize(triage)}";
            var diagnosticsAgent = await agents.CreateDiagnosticsAgentAsync(ct);

            var knowledgeTask = RunAgentAsync(incident, agents.CreateKnowledgeAgent(), context, ct);
            var diagnosticsTask = RunAgentAsync(incident, diagnosticsAgent, context, ct);
            await Task.WhenAll(knowledgeTask, diagnosticsTask);

            // ---------- 3. Resolution ----------
            var resolutionPrompt = $"""
                {context}

                ## Knowledge briefing
                {knowledgeTask.Result}

                ## Diagnostics findings
                {diagnosticsTask.Result}
                """;
            var resolutionRaw = await RunAgentAsync(incident, agents.CreateResolutionAgent(), resolutionPrompt, ct);
            var resolution = Json.ParseLlmJson<ResolutionResult>(resolutionRaw);

            // ---------- 4. Guardrail: action allow-list ----------
            var (action, violation) = ActionPolicy.Enforce(resolution.Action);
            if (violation is not null)
                await RecordAsync(incident, "ActionPolicy", "guardrail", $"Policy applied ({violation}). Final action: {action.ActionType}", ct);

            incident.RootCause = resolution.RootCause;
            incident.Confidence = Math.Clamp(resolution.Confidence, 0, 1);
            incident.Summary = resolution.Summary;
            incident.ProposedActionJson = Json.Serialize(action);

            // ---------- 5. Pause for a human ----------
            await SetStatusAsync(incident, IncidentStatus.AwaitingApproval, ct);
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Investigation failed for {IncidentId}", incidentId);
            activity?.SetStatus(ActivityStatusCode.Error, ex.Message);
            await RecordAsync(incident, "Orchestrator", "error", ex.Message, CancellationToken.None);
            await SetStatusAsync(incident, IncidentStatus.Failed, CancellationToken.None);
        }
    }

    public async Task<Incident?> ApproveAsync(Guid incidentId, DecisionRequest decision, CancellationToken ct)
    {
        var incident = await db.Incidents.FirstOrDefaultAsync(i => i.Id == incidentId, ct);
        if (incident is null || incident.Status != IncidentStatus.AwaitingApproval || incident.ProposedActionJson is null)
            return null;

        incident.DecidedBy = decision.DecidedBy;
        incident.DecisionNote = decision.Note;
        await SetStatusAsync(incident, IncidentStatus.Executing, ct);
        await RecordAsync(incident, "Human", "status", $"Approved by {decision.DecidedBy}", ct);

        try
        {
            var action = Json.Deserialize<ProposedAction>(incident.ProposedActionJson)!;
            incident.ExecutionResult = await remediation.ExecuteAsync(action, ct);
            await RecordAsync(incident, "Executor", "output", incident.ExecutionResult, ct);
            await SetStatusAsync(incident, IncidentStatus.Resolved, ct);
        }
        catch (Exception ex)
        {
            await RecordAsync(incident, "Executor", "error", ex.Message, ct);
            await SetStatusAsync(incident, IncidentStatus.Failed, ct);
        }
        return incident;
    }

    public async Task<Incident?> RejectAsync(Guid incidentId, DecisionRequest decision, CancellationToken ct)
    {
        var incident = await db.Incidents.FirstOrDefaultAsync(i => i.Id == incidentId, ct);
        if (incident is null || incident.Status != IncidentStatus.AwaitingApproval) return null;

        incident.DecidedBy = decision.DecidedBy;
        incident.DecisionNote = decision.Note;
        await RecordAsync(incident, "Human", "status", $"Rejected by {decision.DecidedBy}: {decision.Note}", ct);
        await SetStatusAsync(incident, IncidentStatus.Rejected, ct);
        return incident;
    }

    // ---------------- helpers ----------------

    private static string IncidentPrompt(Incident i) => $"""
        ## Incident report
        Title: {i.Title}
        Description: {i.Description}
        """;

    /// <summary>Streams an agent run, recording tool calls live so the UI timeline updates in real time.</summary>
    private async Task<string> RunAgentAsync(Incident incident, AIAgent agent, string prompt, CancellationToken ct)
    {
        var name = agent.Name ?? "Agent";
        using var activity = Activity.StartActivity($"agent.{name}");
        await RecordAsync(incident, name, "started", "Working…", ct);

        var text = new StringBuilder();
        await foreach (var update in agent.RunStreamingAsync(prompt, cancellationToken: ct))
        {
            foreach (var content in update.Contents)
            {
                switch (content)
                {
                    case FunctionCallContent call:
                        await RecordAsync(incident, name, "tool_call",
                            $"{call.Name}({Json.Serialize(call.Arguments)})", ct);
                        break;
                    case FunctionResultContent result:
                        await RecordAsync(incident, name, "tool_result",
                            Json.Truncate(result.Result?.ToString(), 800), ct);
                        break;
                }
            }
            text.Append(update.Text);
        }

        var output = text.ToString().Trim();
        await RecordAsync(incident, name, "output", Json.Truncate(output, 4000), ct);
        return output;
    }

    private async Task RecordAsync(Incident incident, string agent, string kind, string content, CancellationToken ct)
    {
        var step = new AgentStep { IncidentId = incident.Id, Agent = agent, Kind = kind, Content = content };
        await _dbGate.WaitAsync(ct);
        try
        {
            db.AgentSteps.Add(step);
            await db.SaveChangesAsync(ct);
        }
        finally { _dbGate.Release(); }
        bus.Publish(incident.Id, step.ToDto());
    }

    private async Task SetStatusAsync(Incident incident, IncidentStatus status, CancellationToken ct)
    {
        incident.Status = status;
        incident.UpdatedAt = DateTimeOffset.UtcNow;
        await SaveAsync(ct);
        await RecordAsync(incident, "Orchestrator", "status", status.ToString(), ct);
    }

    private async Task SaveAsync(CancellationToken ct)
    {
        await _dbGate.WaitAsync(ct);
        try { await db.SaveChangesAsync(ct); }
        finally { _dbGate.Release(); }
    }
}

    /// <summary>Uses ConnectionStrings:OpsPilot if present, otherwise Db:Host/Name/Username/Password.</summary>
    private static string BuildConnectionString(IConfiguration config)
    {
        var explicitCs = config.GetConnectionString("OpsPilot");
        if (!string.IsNullOrWhiteSpace(explicitCs)) return explicitCs;

        return new NpgsqlConnectionStringBuilder
        {
            Host = config["Db:Host"] ?? "localhost",
            Port = int.TryParse(config["Db:Port"], out var p) ? p : 5432,
            Database = config["Db:Name"] ?? "opspilot",
            Username = config["Db:Username"] ?? "opspilot",
            Password = config["Db:Password"] ?? "opspilot",
            SslMode = config["Db:Host"] is { } h && h.Contains("rds.amazonaws.com") ? SslMode.Require : SslMode.Prefer
        }.ConnectionString;
    }

    /// <summary>Create schema (incl. pgvector extension) and seed runbooks on first start.</summary>
    public static async Task InitializeOpsPilotAsync(this IServiceProvider sp, CancellationToken ct = default)
    {
        await using (var scope = sp.CreateAsyncScope())
        {
            var db = scope.ServiceProvider.GetRequiredService<OpsPilotDbContext>();
            await db.Database.EnsureCreatedAsync(ct); // switch to EF migrations once the schema stabilises

            var conn = (NpgsqlConnection)db.Database.GetDbConnection();
            await conn.OpenAsync(ct);
            await conn.ReloadTypesAsync(ct);
            await conn.CloseAsync();
        }
        var seedPath = sp.GetRequiredService<IOptions<KnowledgeOptions>>().Value.SeedPath;
        await sp.GetRequiredService<IngestionService>().SeedFromFolderAsync(seedPath, ct);
    }

namespace OpsPilot.Api.Endpoints;

public static class IncidentEndpoints
{
    public static void MapIncidentEndpoints(this IEndpointRouteBuilder app)
    {
        var g = app.MapGroup("/api/incidents").WithTags("Incidents");

        g.MapPost("/", async (CreateIncidentRequest req, OpsPilotDbContext db, IncidentQueue queue, CancellationToken ct) =>
        {
            var (ok, reason) = PromptGuard.ValidateUserInput(req.Title, req.Description);
            if (!ok) return Results.BadRequest(new { error = reason });

            var incident = new Incident { Title = req.Title.Trim(), Description = req.Description.Trim() };
            db.Incidents.Add(incident);
            await db.SaveChangesAsync(ct);
            await queue.EnqueueAsync(incident.Id, ct);
            return Results.Accepted($"/api/incidents/{incident.Id}", new { id = incident.Id });
        });

        g.MapGet("/", async (OpsPilotDbContext db, CancellationToken ct) =>
            await db.Incidents.OrderByDescending(i => i.CreatedAt).Take(50)
                .Select(i => new IncidentSummaryDto(i.Id, i.Title, i.Status, i.Severity, i.Service, i.Category, i.CreatedAt))
                .ToListAsync(ct));

        g.MapGet("/{id:guid}", async (Guid id, OpsPilotDbContext db, CancellationToken ct) =>
        {
            var incident = await db.Incidents.Include(i => i.Steps).AsNoTracking().FirstOrDefaultAsync(i => i.Id == id, ct);
            return incident is null ? Results.NotFound() : Results.Ok(incident.ToDetail());
        });

        // Live agent timeline via Server-Sent Events (.NET 10 TypedResults.ServerSentEvents)
        g.MapGet("/{id:guid}/stream", (Guid id, OpsPilotDbContext db, IncidentEventBus bus, CancellationToken ct) =>
            TypedResults.ServerSentEvents(StreamSteps(id, db, bus, ct)));

        g.MapPost("/{id:guid}/approve", async (Guid id, DecisionRequest req, IncidentOrchestrator orch, CancellationToken ct) =>
        {
            var result = await orch.ApproveAsync(id, req, ct);
            return result is null ? Results.Conflict(new { error = "Incident is not awaiting approval." }) : Results.Ok(new { status = result.Status });
        });

        g.MapPost("/{id:guid}/reject", async (Guid id, DecisionRequest req, IncidentOrchestrator orch, CancellationToken ct) =>
        {
            var result = await orch.RejectAsync(id, req, ct);
            return result is null ? Results.Conflict(new { error = "Incident is not awaiting approval." }) : Results.Ok(new { status = result.Status });
        });
    }

    private static async IAsyncEnumerable<SseItem<AgentStepDto?>> StreamSteps(
        Guid id, OpsPilotDbContext db, IncidentEventBus bus, [EnumeratorCancellation] CancellationToken ct)
    {
        // Subscribe BEFORE replaying history so no step is lost in between; dedupe by id.
        var (token, reader) = bus.Subscribe(id);
        var seen = new HashSet<Guid>();
        try
        {
            var history = await db.AgentSteps.AsNoTracking().Where(s => s.IncidentId == id)
                .OrderBy(s => s.At).ToListAsync(ct);
            foreach (var s in history)
            {
                seen.Add(s.Id);
                yield return new SseItem<AgentStepDto?>(s.ToDto(), "step");
            }

            while (!ct.IsCancellationRequested)
            {
                // Heartbeat every 15s keeps load balancers from closing an idle stream.
                var wait = reader.WaitToReadAsync(ct).AsTask();
                var finished = await Task.WhenAny(wait, Task.Delay(TimeSpan.FromSeconds(15), ct));
                if (finished != wait)
                {
                    yield return new SseItem<AgentStepDto?>(null, "ping");
                    continue;
                }
                if (!await wait) yield break;
                while (reader.TryRead(out var step))
                    if (seen.Add(step.Id))
                        yield return new SseItem<AgentStepDto?>(step, "step");
            }
        }
        finally { bus.Unsubscribe(id, token); }
    }
}

        }).DisableAntiforgery();

            await retrieval.SearchAsync(q, 5, ct));

/// <summary>
/// Deterministic fake production environment with 4 seeded problems, so demos and evals are repeatable:
///  - checkout-api: bad deploy v2.4.1 exhausted the DB connection pool -> 502s (fix: rollback to v2.4.0)
///  - inventory-service: memory leak, OOMKilled restarts (fix: restart_service)
///  - payment-service: traffic spike, CPU saturated (fix: scale_service)
///  - auth-service: TLS cert expiring in 2 days (fix: create_ticket)
/// </summary>
public sealed class FakeOpsState
{
    private readonly object _lock = new();
    private Dictionary<string, ServiceHealth> _health = [];
    private List<Deployment> _deployments = [];
    private List<LogEntry> _logs = [];
    private int _ticketSeq = 1000;

    public FakeOpsState() => Reset();

    public void Reset()
    {
        var now = DateTimeOffset.UtcNow;
        lock (_lock)
        {
            _health = new()
            {
                ["checkout-api"] = new("checkout-api", "degraded", 18.4, 2400, 61, 72, 3, 0, "v2.4.1"),
                ["payment-service"] = new("payment-service", "degraded", 2.1, 1900, 48, 97, 2, 0, "v5.2.0"),
                ["inventory-service"] = new("inventory-service", "unhealthy", 6.5, 800, 96, 40, 2, 4, "v3.1.2"),
                ["auth-service"] = new("auth-service", "healthy", 0.1, 120, 35, 20, 3, 0, "v1.9.0"),
            };
            _deployments =
            [
                new("checkout-api", "v2.4.1", now.AddMinutes(-35), "ci-bot", "Refactor order repository; new DB query per cart item"),
                new("checkout-api", "v2.4.0", now.AddDays(-4), "ci-bot", "Stable release"),
                new("payment-service", "v5.2.0", now.AddDays(-9), "ci-bot", "Stable release"),
                new("inventory-service", "v3.1.2", now.AddDays(-3), "ci-bot", "Add in-memory product cache"),
                new("auth-service", "v1.9.0", now.AddDays(-20), "ci-bot", "Stable release"),
            ];
            _logs =
            [
                new(now.AddMinutes(-30), "checkout-api", "ERROR", "Npgsql.NpgsqlException: The connection pool has been exhausted (Max Pool Size=20)"),
                new(now.AddMinutes(-28), "checkout-api", "ERROR", "Upstream request failed: 502 Bad Gateway returned to client for POST /api/checkout"),
                new(now.AddMinutes(-12), "checkout-api", "ERROR", "Timeout acquiring DB connection after 15000ms in OrderRepository.GetCartItems"),
                new(now.AddMinutes(-20), "payment-service", "WARN", "Thread pool starvation detected; queue length 4812"),
                new(now.AddMinutes(-8), "payment-service", "ERROR", "Request timeout after 2000ms on POST /payments/authorize (CPU 97%)"),
                new(now.AddMinutes(-40), "inventory-service", "ERROR", "Container OOMKilled (memory limit 1024Mi)"),
                new(now.AddMinutes(-15), "inventory-service", "ERROR", "Container OOMKilled (memory limit 1024Mi); ProductCache entries=2.1M"),
                new(now.AddMinutes(-60), "auth-service", "WARN", "TLS certificate for auth.internal expires in 2 days"),
            ];
        }
    }

    public IReadOnlyList<string> Services() { lock (_lock) return _health.Keys.ToList(); }

    public ServiceHealth? Health(string service) { lock (_lock) return _health.GetValueOrDefault(service); }

    public IReadOnlyList<LogEntry> Logs(string service, string? level, int minutes)
    {
        var since = DateTimeOffset.UtcNow.AddMinutes(-minutes);
        lock (_lock)
            return _logs.Where(l => l.Service == service && l.Timestamp >= since
                                    && (level is null || l.Level.Equals(level, StringComparison.OrdinalIgnoreCase)))
                        .OrderByDescending(l => l.Timestamp).ToList();
    }

    public IReadOnlyList<Deployment> Deployments(string service)
    {
        lock (_lock) return _deployments.Where(d => d.Service == service).OrderByDescending(d => d.DeployedAt).ToList();
    }

    public RemediationResult Apply(RemediationRequest req)
    {
        lock (_lock)
        {
            if (req.ActionType is "create_ticket" or "no_action")
            {
                var id = req.ActionType == "create_ticket" ? $"OPS-{++_ticketSeq}" : null;
                return new(true, id is null ? "No action taken." : $"Ticket {id} created for {req.Service}.", id);
            }
            if (!_health.TryGetValue(req.Service, out var h))
                return new(false, $"Unknown service {req.Service}");

            switch (req.ActionType)
            {
                case "rollback_deployment":
                    var to = req.Parameters?.GetValueOrDefault("toVersion");
                    if (to is null || !_deployments.Any(d => d.Service == req.Service && d.Version == to))
                        return new(false, $"Version '{to}' not found for {req.Service}");
                    _health[req.Service] = h with { Status = "healthy", ErrorRatePercent = 0.3, P95LatencyMs = 180, CurrentVersion = to };
                    _deployments.Insert(0, new(req.Service, to, DateTimeOffset.UtcNow, "opspilot", "Rollback approved by on-call"));
                    return new(true, $"Rolled back {req.Service} to {to}. Error rate now 0.3%.");

                case "restart_service":
                    _health[req.Service] = h with { Status = "healthy", MemoryPercent = 42, RestartsLastHour = 0, ErrorRatePercent = 0.2 };
                    return new(true, $"Rolling restart of {req.Service} completed. Memory at 42%.");

                case "scale_service":
                    var replicas = int.TryParse(req.Parameters?.GetValueOrDefault("replicas"), out var r) ? r : h.Replicas + 2;
                    _health[req.Service] = h with { Replicas = replicas, CpuPercent = Math.Max(30, h.CpuPercent * h.Replicas / replicas),
                        Status = "healthy", P95LatencyMs = 250, ErrorRatePercent = 0.2 };
                    return new(true, $"Scaled {req.Service} to {replicas} replicas.");

                default:
                    return new(false, $"Unsupported action {req.ActionType}");
            }
        }
    }
}

using System.Diagnostics;
using System.Net.Http.Json;
using System.Text.Json;

// End-to-end eval: submits each case to the running API, waits for the agents to finish,
// and scores service identification + proposed action. Run: dotnet run -- http://localhost:5080
var baseUrl = args.FirstOrDefault() ?? "http://localhost:5080";
var http = new HttpClient { BaseAddress = new Uri(baseUrl), Timeout = TimeSpan.FromMinutes(5) };
var json = new JsonSerializerOptions(JsonSerializerDefaults.Web);
var cases = JsonSerializer.Deserialize<List<EvalCase>>(await File.ReadAllTextAsync("cases.json"), json)!;

// Best-effort reset of the fake environment
try { await http.PostAsync("http://localhost:8081/admin/reset", null); } catch { }

var rows = new List<string>();
int serviceHits = 0, actionHits = 0;

foreach (var c in cases)
{
    var sw = Stopwatch.StartNew();
    var resp = await http.PostAsJsonAsync("/api/incidents", new { c.Title, c.Description });

    if (c.ExpectedAction == "REJECT")
    {
        var rejected = resp.StatusCode == System.Net.HttpStatusCode.BadRequest;
        if (rejected) { serviceHits++; actionHits++; }
        rows.Add($"| {c.Title} | REJECT | {(rejected ? "rejected" : "accepted")} | {(rejected ? "✅" : "❌")} | {(rejected ? "✅" : "❌")} | {sw.ElapsedMilliseconds} ms |");
        continue;
    }

    var id = (await resp.Content.ReadFromJsonAsync<JsonElement>(json)).GetProperty("id").GetString();
    JsonElement detail;
    do
    {
        await Task.Delay(2000);
        detail = await http.GetFromJsonAsync<JsonElement>($"/api/incidents/{id}", json);
    } while (detail.GetProperty("status").GetString() is "Received" or "Triaging" or "Investigating"
             && sw.Elapsed < TimeSpan.FromMinutes(4));

    var service = detail.TryGetProperty("service", out var s) ? s.GetString() : null;
    var action = detail.TryGetProperty("proposedAction", out var a) && a.ValueKind == JsonValueKind.Object
        ? a.GetProperty("actionType").GetString() : null;

    var sOk = service == c.ExpectedService; if (sOk) serviceHits++;
    var aOk = action == c.ExpectedAction; if (aOk) actionHits++;
    rows.Add($"| {c.Title} | {c.ExpectedService} / {c.ExpectedAction} | {service} / {action} | {(sOk ? "✅" : "❌")} | {(aOk ? "✅" : "❌")} | {sw.ElapsedMilliseconds} ms |");
}

Console.WriteLine("| Case | Expected | Actual | Service | Action | Latency |");
Console.WriteLine("|---|---|---|---|---|---|");
rows.ForEach(Console.WriteLine);
Console.WriteLine($"\nService accuracy: {serviceHits}/{cases.Count}  ·  Action accuracy: {actionHits}/{cases.Count}");

# Runbook: checkout-api returning 502 / high error rate

## Symptoms
- Customers see "Something went wrong" at checkout; load balancer returns 502 Bad Gateway.
- checkout-api error rate above 5%, p95 latency above 1500 ms.

## Likely causes
1. A recent deployment introduced a regression (most common, check deployments in the last 2 hours).
2. Database connection pool exhaustion ("connection pool has been exhausted"), usually caused by N+1 queries or connections not being disposed.
3. Downstream payment-service timeouts.

## Diagnosis steps
1. Check service health for checkout-api (error rate, latency, version).
2. Search ERROR logs for the last 60 minutes. Look for Npgsql pool exhaustion or upstream timeouts.
3. List recent deployments. If a deployment happened shortly before errors began, treat it as the prime suspect.

## Remediation
- If a deployment correlates with the start of errors: **roll back to the previous stable version** (rollback_deployment with toVersion = previous version). Risk: low.
- If pool exhaustion without a recent deploy: scale replicas temporarily and open a ticket for the owning team.
- Never restart all replicas at once during peak traffic.

# Runbook: Service restarting with OOMKilled

## Symptoms
- Container restarts repeatedly; logs show "OOMKilled".
- Memory usage above 90% of the container limit.

## Likely causes
1. Unbounded in-memory cache (no eviction / size limit).
2. Memory leak introduced by a recent change.
3. Memory limit set too low for current traffic.

## Diagnosis steps
1. Check service health: memory %, restart count in the last hour.
2. Search ERROR logs for OOMKilled and cache-size messages.
3. Review the most recent deployment notes for caching or memory-related changes.

## Remediation
- Immediate mitigation: **rolling restart of the service** (restart_service). Risk: low. Buys time but does not fix a leak.
- Follow-up: open a ticket to add cache eviction (e.g. MemoryCache SizeLimit) to the owning team.
- Do not roll back if the deployment is older than 48 hours unless a restart fails to stabilise memory.

# Runbook: CPU saturation and request timeouts

## Symptoms
- CPU above 90% across replicas, p95 latency rising, request timeouts.
- Logs show thread pool starvation or long request queues.

## Likely causes
1. Traffic spike (sales event, marketing campaign, retry storm from an upstream).
2. Inefficient code path from a recent deployment.

## Diagnosis steps
1. Check service health: CPU %, replicas, latency.
2. Search WARN and ERROR logs for thread pool starvation or timeouts.
3. Check deployments: if no recent deployment, a traffic spike is the most likely cause.

## Remediation
- **Scale out** (scale_service) to roughly double the replicas, capped at 10. Risk: low.
- If a recent deployment correlates, prefer rollback over scaling.
- Open a ticket to review autoscaling thresholds.

# Past incident reports

## INC-2041: checkout-api 502s after release v2.2.0
A release added a per-item database query in the cart flow (N+1 pattern). Under load the Npgsql connection pool (max 20) was exhausted and checkout-api returned 502s at ~15% error rate. Resolved by rolling back to v2.1.3 within 10 minutes. Follow-up: batch the query, add load test to CI.

import { HttpClient } from '@angular/common/http';
import { Injectable, NgZone, inject } from '@angular/core';
import { Observable } from 'rxjs';
import { AgentStep, IncidentDetail, IncidentSummary, KnowledgeSource } from './models';

@Injectable({ providedIn: 'root' })
export class IncidentService {
  private http = inject(HttpClient);
  private zone = inject(NgZone);

  list() { return this.http.get<IncidentSummary[]>('/api/incidents'); }
  get(id: string) { return this.http.get<IncidentDetail>(`/api/incidents/${id}`); }
  create(title: string, description: string) {
    return this.http.post<{ id: string }>('/api/incidents', { title, description });
  }
  approve(id: string, decidedBy: string, note?: string) {
    return this.http.post(`/api/incidents/${id}/approve`, { decidedBy, note });
  }
  reject(id: string, decidedBy: string, note?: string) {
    return this.http.post(`/api/incidents/${id}/reject`, { decidedBy, note });
  }

  /** Live agent timeline over Server-Sent Events. The browser reconnects automatically; the API replays history. */
  stream(id: string): Observable<AgentStep> {
    return new Observable<AgentStep>(subscriber => {
      const es = new EventSource(`/api/incidents/${id}/stream`);
      es.addEventListener('step', e =>
        this.zone.run(() => subscriber.next(JSON.parse((e as MessageEvent).data) as AgentStep)));
      return () => es.close();
    });
  }

  sources() { return this.http.get<KnowledgeSource[]>('/api/knowledge/sources'); }
  upload(file: File) {
    const form = new FormData();
    form.append('file', file);
    return this.http.post<{ source: string; chunks: number }>('/api/knowledge', form);
  }
}

export class IncidentDetailPage implements OnInit, OnDestroy {
  /** Bound from the :id route param via withComponentInputBinding(). */
  id = input.required<string>();

  private api = inject(IncidentService);
  private sub?: Subscription;

  incident = signal<IncidentDetail | null>(null);
  steps = signal<AgentStep[]>([]);
  loadError = signal<string | null>(null);
  formError = signal<string | null>(null);
  busy = signal(false);
  decidedBy = '';
  note = '';

  visibleSteps = computed(() => this.steps().filter(s => s.kind !== 'started'));
  working = computed(() => ['Received', 'Triaging', 'Investigating', 'Executing'].includes(this.incident()?.status ?? ''));
  panelTitle = computed(() => ({
    AwaitingApproval: 'Your decision is needed',
    Resolved: 'Resolved', Rejected: 'Fix rejected', Failed: 'Investigation failed',
    Executing: 'Running the approved fix'
  } as Record<string, string>)[this.incident()?.status ?? ''] ?? 'Investigation in progress');

  ngOnInit() {
    this.refresh();
    this.sub = this.api.stream(this.id()).subscribe(step => {
      this.steps.update(list => list.some(s => s.id === step.id) ? list : [...list, step]);
      if (step.kind === 'status' || step.agent === 'TriageAgent') this.refresh(); // pull updated fields
    });
  }
  ngOnDestroy() { this.sub?.unsubscribe(); }

  refresh() {
    this.api.get(this.id()).subscribe({
      next: d => {
        this.incident.set(d);
        this.steps.update(list => {
          const ids = new Set(list.map(s => s.id));
          return [...list, ...d.steps.filter(s => !ids.has(s.id))].sort((a, b) => a.at.localeCompare(b.at));
        });
      },
      error: () => this.loadError.set('This incident could not be loaded. It may have been deleted, or the API is down.')
    });
  }

  approve() { this.decide('approve'); }
  reject() { this.decide('reject'); }

  private decide(kind: 'approve' | 'reject') {
    if (!this.decidedBy.trim()) { this.formError.set('Enter your name so the decision is attributed.'); return; }
    this.formError.set(null);
    this.busy.set(true);
    const call = kind === 'approve'
      ? this.api.approve(this.id(), this.decidedBy, this.note)
      : this.api.reject(this.id(), this.decidedBy, this.note);
    call.subscribe({
      next: () => { this.busy.set(false); this.refresh(); },
      error: () => { this.busy.set(false); this.formError.set('The decision was not saved. Refresh and try again.'); }
    });
  }

  statusText(s: string) {
    return ({ AwaitingApproval: 'Waiting for human approval', Triaging: 'Triage started',
              Investigating: 'Knowledge and diagnostics agents running in parallel' } as Record<string, string>)[s] ?? s;
  }
}

Quick checks if something fails:

npx @modelcontextprotocol/inspector lets you test the MCP tools on their own at http://localhost:8081/mcp.

The .NET code hasn’t been compiled yet, so expect a few small errors on the first build from library version differences. The most likely ones:

Older Agent Framework versions use CreateAIAgent instead of AsAIAgent.
Older MCP SDK versions use McpClientFactory.CreateAsync and SseClientTransport instead of McpClient.CreateAsync and HttpClientTransport.
