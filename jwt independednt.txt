The fix is to give each user two independent identities, which also matches how real companies work:

Identity	What it is	Where it’s used	Who can reset it
HelpBot portal account	The app’s own user database (SQLite, hashed passwords)	Logging in to the helpdesk app (/login, /it/login)	Nobody needs to; it’s separate
Directory account (Keycloak)	The “corporate account” the helpdesk manages	Keycloak Account Console	The AI agent (after IT approval), or IT directly from the dashboard

Why your login broke: in the previous version the app login was the Keycloak login. Every wrong attempt in the app also counted toward Keycloak’s 3-attempt lockout, and the app’s change-password page rewrote the Keycloak user. Trying the temporary and new passwords a few times almost certainly locked the account, after which even the correct password fails. In the new design the app never touches Keycloak passwords during login, and Keycloak’s own screen handles the “set a new password” step after a reset.

using Helpdesk.Application.Common;
using Helpdesk.Application.Services;
using Microsoft.EntityFrameworkCore;

namespace Helpdesk.Application.Auth;

public record LoginRequest(string Username, string Password, string Portal);
public record LoginResult(string AccessToken, DateTime ExpiresAt, string Username, string FullName,
    string Role, string Portal, string? DirectoryAccount);

/// Portal login against the app's OWN user store. Completely independent of Keycloak.
public class AuthService(IAppDbContext db, IAppPasswordHasher hasher, ITokenService tokens,
    ITokenRevocationStore revocations, AuditLogger audit)
{
    private const int MaxFailedAttempts = 5;
    private static readonly TimeSpan LockoutDuration = TimeSpan.FromMinutes(5);

    public async Task<LoginResult> LoginAsync(LoginRequest request, CancellationToken ct = default)
    {
        var username = (request.Username ?? "").Trim().ToLowerInvariant();
        var portal = (request.Portal ?? "").Trim().ToLowerInvariant();

        if (username.Length == 0 || string.IsNullOrEmpty(request.Password))
            throw new AuthenticationFailedException("Username and password are required.");
        if (portal is not (Portals.Employee or Portals.It))
            throw new InvalidOperationException("Unknown portal.");

        var user = await db.AppUsers.FirstOrDefaultAsync(u => u.Username == username, ct);
        if (user is null)
        {
            await audit.LogAsync(username, "portal_login", $"portal={portal}", "Failed-UnknownUser", ct);
            throw new AuthenticationFailedException("Invalid username or password.");
        }

        if (user.LockoutEndUtc > DateTime.UtcNow)
        {
            await audit.LogAsync(username, "portal_login", $"portal={portal}", "Failed-Locked", ct);
            var minutes = Math.Ceiling((user.LockoutEndUtc.Value - DateTime.UtcNow).TotalMinutes);
            throw new AuthenticationFailedException($"Too many failed attempts. Try again in {minutes} minute(s).");
        }

        if (!hasher.Verify(user, request.Password))
        {
            user.FailedLoginCount++;
            if (user.FailedLoginCount >= MaxFailedAttempts)
            {
                user.LockoutEndUtc = DateTime.UtcNow.Add(LockoutDuration);
                user.FailedLoginCount = 0;
            }
            await db.SaveChangesAsync(ct);
            await audit.LogAsync(username, "portal_login", $"portal={portal}", "Failed-BadPassword", ct);
            throw new AuthenticationFailedException("Invalid username or password.");
        }

        user.FailedLoginCount = 0;
        user.LockoutEndUtc = null;
        await db.SaveChangesAsync(ct);

        // Portal separation: employees -> chat portal, IT approvers -> IT portal
        var requiredRole = portal == Portals.It ? Roles.Approver : Roles.Employee;
        if (user.Role != requiredRole)
        {
            await audit.LogAsync(username, "portal_login", $"portal={portal}", "Denied-WrongPortal", ct);
            throw new ForbiddenAccessException(portal == Portals.It
                ? "This account is not authorized for the IT portal."
                : "IT staff must sign in through the IT portal.");
        }

        var issued = tokens.CreateToken(user, portal);
        await audit.LogAsync(username, "portal_login", $"portal={portal}, jti={issued.Jti}", "Success", ct);

        return new LoginResult(issued.Token, issued.ExpiresAt, user.Username, user.FullName,
            user.Role, portal, user.DirectoryAccount);
    }

    public async Task LogoutAsync(string username, string jti, DateTime expiresAt, CancellationToken ct = default)
    {
        revocations.Revoke(jti, expiresAt);
        await audit.LogAsync(username, "portal_logout", $"jti={jti}", "Success", ct);
    }
}

using System.Text.Json;
using Helpdesk.Application.Common;
using Helpdesk.Domain;
using Microsoft.Extensions.Options;

namespace Helpdesk.Application.Services;

/// Deterministic code that performs high-risk actions ONLY after human approval.
public class ActionExecutor(IIdentityService directory, IServerService servers, IEmailService email,
    IOptions<HelpdeskOptions> options)
{
    public const string ResetPassword = "reset_my_password";
    public const string RestartServer = "restart_server";

    public async Task<string?> ValidateAsync(string action, string? directoryAccount,
        IDictionary<string, string> args, CancellationToken ct)
    {
        switch (action)
        {
            case ResetPassword:
                if (string.IsNullOrEmpty(directoryAccount))
                    return "You have no linked directory (Keycloak) account.";
                return await directory.GetUserAsync(directoryAccount, ct) is null
                    ? $"Directory account '{directoryAccount}' was not found." : null;
            case RestartServer:
                var name = args.TryGetValue("serverName", out var n) ? n : "";
                return await servers.GetAsync(name, ct) is null
                    ? $"'{name}' is not a managed server. Use list_servers to see valid names." : null;
            default:
                return $"Action '{action}' is not supported by the approval workflow.";
        }
    }

    public async Task<string> ExecuteAsync(ApprovalRequest req, CancellationToken ct)
    {
        var args = JsonSerializer.Deserialize<Dictionary<string, string>>(req.ArgumentsJson) ?? new();

        switch (req.ActionName)
        {
            case ResetPassword:
                return await ResetAndNotifyAsync(req.DirectoryAccount, $"Approval request {req.Id}", ct);
            case RestartServer:
            {
                var name = args["serverName"];
                await servers.RestartAsync(name, ct);
                var s = await servers.GetAsync(name, ct);
                return $"Server '{name}' was restarted. Current state: {s?.State ?? "unknown"}.";
            }
            default:
                throw new InvalidOperationException($"Unsupported action '{req.ActionName}'.");
        }
    }

    /// Resets the Keycloak password to a temporary one and emails it. Keycloak forces a new password at next login.
    public async Task<string> ResetAndNotifyAsync(string directoryAccount, string reference, CancellationToken ct)
    {
        var user = await directory.GetUserAsync(directoryAccount, ct)
                   ?? throw new InvalidOperationException("Directory account no longer exists.");
        var temp = await directory.ResetPasswordAsync(directoryAccount, ct);

        await email.SendAsync(user.Email, "Your temporary directory password",
            $"Hello {user.FullName},\n\n" +
            "Your directory (Keycloak) password was reset.\n" +
            $"Temporary password: {temp}\n\n" +
            $"Sign in here: {options.Value.AccountConsoleUrl}\n" +
            "You will be asked to choose a new password immediately.\n\n" +
            "Note: this does NOT change your HelpBot portal password.\n\n" +
            $"Reference: {reference}", ct);

        return $"Directory password reset. A temporary password was emailed to {Mask(user.Email)}.";
    }

    private static string Mask(string email)
    {
        var at = email.IndexOf('@');
        return at <= 1 ? "***" + email[Math.Max(at, 0)..] : email[0] + "***" + email[at..];
    }
}

using System.Text.Json;
using Helpdesk.Application.Common;
using Helpdesk.Domain;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Options;

namespace Helpdesk.Application.Services;

public class ApprovalService(IAppDbContext db, ActionExecutor executor, AuditLogger audit,
    IOptions<HelpdeskOptions> options)
{
    public static string Short(Guid id) => id.ToString()[..8];

    public async Task<string> RequestAsync(string requester, string? directoryAccount, string action,
        Dictionary<string, string> args, RiskLevel risk, CancellationToken ct = default)
    {
        var error = await executor.ValidateAsync(action, directoryAccount, args, ct);
        if (error is not null)
        {
            await audit.LogAsync(requester, action, error, "Rejected-Validation", ct);
            return $"Cannot request this action: {error}";
        }

        var argsJson = JsonSerializer.Serialize(args);

        var existing = await db.Approvals.FirstOrDefaultAsync(a =>
            a.Requester == requester && a.ActionName == action &&
            a.ArgumentsJson == argsJson && a.Status == ApprovalStatus.Pending, ct);
        if (existing is not null)
            return $"APPROVAL_PENDING: An identical request ({Short(existing.Id)}) is already waiting for IT approval.";

        var req = new ApprovalRequest
        {
            Requester = requester,
            DirectoryAccount = directoryAccount ?? "",
            ActionName = action,
            ArgumentsJson = argsJson,
            Risk = risk,
            Reason = args.GetValueOrDefault("reason") ?? "",
            ExpiresAt = DateTime.UtcNow.AddMinutes(options.Value.ApprovalExpiryMinutes)
        };
        db.Approvals.Add(req);
        await db.SaveChangesAsync(ct);
        await audit.LogAsync(requester, action, $"Approval {req.Id} created. Args: {argsJson}", "PendingApproval", ct);

        return $"APPROVAL_REQUIRED: Request {Short(req.Id)} was created and is waiting for an IT administrator. " +
               "The action has NOT been performed yet.";
    }

    public async Task<List<ApprovalRequest>> ListAsync(CancellationToken ct = default)
    {
        await ExpireStaleAsync(ct);
        return await db.Approvals.OrderByDescending(a => a.CreatedAt).Take(100).ToListAsync(ct);
    }

    public async Task<List<ApprovalRequest>> ListForUserAsync(string username, CancellationToken ct = default)
    {
        await ExpireStaleAsync(ct);
        return await db.Approvals.Where(a => a.Requester == username)
            .OrderByDescending(a => a.CreatedAt).Take(10).ToListAsync(ct);
    }

    public async Task<ApprovalRequest> DecideAsync(Guid id, string approver, bool approve, string? note,
        CancellationToken ct = default)
    {
        await ExpireStaleAsync(ct);
        var req = await db.Approvals.FirstOrDefaultAsync(a => a.Id == id, ct)
                  ?? throw new KeyNotFoundException("Approval request not found.");

        if (req.Status != ApprovalStatus.Pending)
            throw new InvalidOperationException($"Request is already {req.Status}.");
        if (string.Equals(req.Requester, approver, StringComparison.OrdinalIgnoreCase))
            throw new ForbiddenAccessException("Separation of duties: you cannot approve your own request.");
        if (!approve && string.IsNullOrWhiteSpace(note))
            throw new InvalidOperationException("A note is required when rejecting a request.");

        req.Approver = approver;
        req.DecisionNote = note;
        req.DecidedAt = DateTime.UtcNow;

        if (!approve)
        {
            req.Status = ApprovalStatus.Rejected;
            req.Result = "Rejected by IT.";
        }
        else
        {
            try
            {
                req.Result = await executor.ExecuteAsync(req, ct);
                req.Status = ApprovalStatus.Executed;
            }
            catch (Exception ex)
            {
                req.Status = ApprovalStatus.Failed;
                req.Result = "Execution failed: " + ex.Message;
            }
        }

        db.ChatMessages.Add(new ChatMessage { Username = req.Requester, Role = "assistant", Content = Notification(req) });
        await db.SaveChangesAsync(ct);
        await audit.LogAsync(approver, $"decision:{req.ActionName}",
            $"Request {req.Id} -> {req.Status}. Note: {note}", req.Status.ToString(), ct);
        return req;
    }

    private async Task ExpireStaleAsync(CancellationToken ct)
    {
        var now = DateTime.UtcNow;
        var stale = await db.Approvals
            .Where(a => a.Status == ApprovalStatus.Pending && a.ExpiresAt < now).ToListAsync(ct);
        if (stale.Count == 0) return;
        foreach (var a in stale) { a.Status = ApprovalStatus.Expired; a.Result = "Expired without a decision."; }
        await db.SaveChangesAsync(ct);
    }

    private static string Notification(ApprovalRequest r) => r.Status switch
    {
        ApprovalStatus.Executed => $"Update on request {Short(r.Id)}: approved by IT. {r.Result}",
        ApprovalStatus.Rejected => $"Update on request {Short(r.Id)}: rejected by IT. Reason: {r.DecisionNote}",
        _ => $"Update on request {Short(r.Id)}: approved, but execution failed. IT has been notified."
    };
}

using Helpdesk.Application.Common;

namespace Helpdesk.Application.Services;

/// Direct directory management for IT staff (the human IS the approver here). Every action needs a reason and is audited.
public class DirectoryAdminService(IIdentityService directory, ActionExecutor executor, AuditLogger audit, ICurrentUser currentUser)
{
    public Task<IReadOnlyList<DirectoryUser>> ListAsync(CancellationToken ct = default)
        => directory.ListUsersAsync(ct);

    public async Task<string> UnlockAsync(string username, string? reason, CancellationToken ct = default)
    {
        RequireReason(reason);
        var user = await directory.GetUserAsync(username, ct)
                   ?? throw new KeyNotFoundException("Directory account not found.");
        if (!user.Locked) return $"{username} is not locked.";

        await directory.UnlockAsync(username, ct);
        await audit.LogAsync(currentUser.Username, "it_unlock", $"account={username}; reason={reason}", "Executed", ct);
        return $"{username} has been unlocked.";
    }

    public async Task<string> ResetPasswordAsync(string username, string? reason, CancellationToken ct = default)
    {
        RequireReason(reason);
        _ = await directory.GetUserAsync(username, ct) ?? throw new KeyNotFoundException("Directory account not found.");

        var result = await executor.ResetAndNotifyAsync(username, $"Direct reset by IT ({currentUser.Username})", ct);
        await audit.LogAsync(currentUser.Username, "it_reset_password", $"account={username}; reason={reason}", "Executed", ct);
        return result;
    }

    private static void RequireReason(string? reason)
    {
        if (string.IsNullOrWhiteSpace(reason) || reason.Trim().Length < 5)
            throw new InvalidOperationException("A reason (at least 5 characters) is required for the audit log.");
    }
}

using System.Text.Json;
using Helpdesk.Application.Common;
using Helpdesk.Application.Guardrails;
using Helpdesk.Application.Services;
using Microsoft.SemanticKernel;

namespace Helpdesk.Application.Agent;

/// Intercepts EVERY tool call the LLM makes. Enforced in code, so it cannot be prompt-injected away.
public class ApprovalGateFilter(ICurrentUser user, ApprovalService approvals, AuditLogger audit)
    : IAutoFunctionInvocationFilter
{
    private const int MaxModelRoundTrips = 5;

    public async Task OnAutoFunctionInvocationAsync(AutoFunctionInvocationContext context,
        Func<AutoFunctionInvocationContext, Task> next)
    {
        var fn = context.Function.Name;
        var args = context.Arguments?.ToDictionary(a => a.Key, a => a.Value?.ToString() ?? "")
                   ?? new Dictionary<string, string>();
        var argsJson = JsonSerializer.Serialize(args);

        if (context.RequestSequenceIndex >= MaxModelRoundTrips)
        {
            context.Result = new FunctionResult(context.Function, "Tool call limit reached for this message.");
            context.Terminate = true;
            return;
        }

        if (!RiskPolicy.IsKnown(fn))
        {
            await audit.LogAsync(user.Username, fn, argsJson, "Blocked-UnknownTool");
            context.Result = new FunctionResult(context.Function, "BLOCKED: this tool is not on the allow-list.");
            return;
        }

        var risk = RiskPolicy.Get(fn);
        if (RiskPolicy.RequiresApproval(risk))
        {
            var message = await approvals.RequestAsync(user.Username, user.DirectoryAccount, fn, args, risk);
            context.Result = new FunctionResult(context.Function, message);
            return;
        }

        try
        {
            await next(context);
            await audit.LogAsync(user.Username, fn, argsJson, "Executed");
        }
        catch (Exception ex)
        {
            await audit.LogAsync(user.Username, fn, $"{argsJson} | {ex.Message}", "Error");
            context.Result = new FunctionResult(context.Function, $"ERROR: {ex.Message}");
        }
    }
}

/// Issues the portal's own JWT (HS256).
public class JwtTokenService(IOptions<JwtOptions> options) : ITokenService
{
    public IssuedToken CreateToken(AppUser user, string portal)
    {
        var o = options.Value;
        var now = DateTime.UtcNow;
        var expires = now.AddMinutes(o.ExpiryMinutes);
        var jti = Guid.NewGuid().ToString("N");

        var claims = new List<Claim>
        {
            new("sub", user.Username),
            new("name", user.FullName),
            new("role", user.Role),
            new("portal", portal),
            new("jti", jti)
        };
        if (!string.IsNullOrEmpty(user.DirectoryAccount))
            claims.Add(new Claim("dir", user.DirectoryAccount)); // linked Keycloak account

        var descriptor = new SecurityTokenDescriptor
        {
            Issuer = o.Issuer,
            Audience = o.Audience,
            IssuedAt = now,
            NotBefore = now,
            Expires = expires,
            Subject = new ClaimsIdentity(claims),
            SigningCredentials = new SigningCredentials(
                new SymmetricSecurityKey(Encoding.UTF8.GetBytes(o.SigningKey)),
                SecurityAlgorithms.HmacSha256)
        };

        return new IssuedToken(new JsonWebTokenHandler().CreateToken(descriptor), jti, expires);
    }

    private static readonly SemaphoreSlim TokenLock = new(1, 1);

    private KeycloakOptions O => options.Value;

    public async Task<IReadOnlyList<DirectoryUser>> ListUsersAsync(CancellationToken ct = default)
    {
        using var res = await http.SendAsync(await BuildAsync(HttpMethod.Get, "users?max=100", ct), ct);
        res.EnsureSuccessStatusCode();
        var arr = await res.Content.ReadFromJsonAsync<JsonElement>(ct);

        var users = new List<DirectoryUser>();
        foreach (var u in arr.EnumerateArray())
        {
            var username = u.GetProperty("username").GetString() ?? "";
            if (username.StartsWith("service-account-")) continue;
            users.Add(await ToDirectoryUserAsync(u, ct));
        }
        return users.OrderBy(u => u.Username).ToList();
    }

    public async Task<DirectoryUser?> GetUserAsync(string username, CancellationToken ct = default)
    {
        var user = await FindUserAsync(username, ct);
        return user is null ? null : await ToDirectoryUserAsync(user.Value, ct);
    }

    public async Task UnlockAsync(string username, CancellationToken ct = default)
    {
        var id = await GetIdAsync(username, ct);
        using var res = await http.SendAsync(await BuildAsync(HttpMethod.Delete, $"attack-detection/brute-force/users/{id}", ct), ct);
        res.EnsureSuccessStatusCode();
    }

    public async Task<string> ResetPasswordAsync(string username, CancellationToken ct = default)
    {
        var id = await GetIdAsync(username, ct);
        var temp = RandomNumberGenerator.GetString(
            "ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnpqrstuvwxyz23456789!@#$%", 14);

        // temporary=true -> Keycloak itself forces "Update password" on the next Keycloak login
        var body = new { type = "password", value = temp, temporary = true };
        using var res = await http.SendAsync(await BuildAsync(HttpMethod.Put, $"users/{id}/reset-password", ct, body), ct);
        res.EnsureSuccessStatusCode();

        // Clear any lockout so the temporary password works immediately
        using var unlock = await http.SendAsync(await BuildAsync(HttpMethod.Delete, $"attack-detection/brute-force/users/{id}", ct), ct);
        return temp;
    }

    // ---------- helpers ----------
    private async Task<DirectoryUser> ToDirectoryUserAsync(JsonElement u, CancellationToken ct)
    {
        var id = u.GetProperty("id").GetString();
        using var res = await http.SendAsync(await BuildAsync(HttpMethod.Get, $"attack-detection/brute-force/users/{id}", ct), ct);
        res.EnsureSuccessStatusCode();
        var bf = await res.Content.ReadFromJsonAsync<JsonElement>(ct);

        var first = u.TryGetProperty("firstName", out var f) ? f.GetString() : "";
        var last = u.TryGetProperty("lastName", out var l) ? l.GetString() : "";

        return new DirectoryUser(
            u.GetProperty("username").GetString() ?? "",
            $"{first} {last}".Trim(),
            u.TryGetProperty("email", out var e) ? e.GetString() ?? "" : "",
            u.GetProperty("enabled").GetBoolean(),
            bf.TryGetProperty("disabled", out var d) && d.GetBoolean(),
            bf.TryGetProperty("numFailures", out var n) ? n.GetInt32() : 0);
    }

    private async Task<string> GetIdAsync(string username, CancellationToken ct)
        => (await FindUserAsync(username, ct))?.GetProperty("id").GetString()
           ?? throw new InvalidOperationException($"Directory account '{username}' not found.");

    private async Task<JsonElement?> FindUserAsync(string username, CancellationToken ct)
    {
        using var res = await http.SendAsync(await BuildAsync(HttpMethod.Get,
            $"users?username={Uri.EscapeDataString(username)}&exact=true", ct), ct);
        res.EnsureSuccessStatusCode();
        var arr = await res.Content.ReadFromJsonAsync<JsonElement>(ct);
        return arr.GetArrayLength() > 0 ? arr[0] : (JsonElement?)null;
    }

    private async Task<HttpRequestMessage> BuildAsync(HttpMethod method, string path, CancellationToken ct, object? body = null)
    {
        var req = new HttpRequestMessage(method, $"{O.BaseUrl}/admin/realms/{O.Realm}/{path}");
        req.Headers.Authorization = new AuthenticationHeaderValue("Bearer", await GetServiceTokenAsync(ct));
        if (body is not null) req.Content = JsonContent.Create(body);
        return req;
    }

    private async Task<string> GetServiceTokenAsync(CancellationToken ct)
    {
        if (_cachedToken is not null && DateTime.UtcNow < _tokenExpiresAt) return _cachedToken;

        await TokenLock.WaitAsync(ct);
        try
        {
            if (_cachedToken is not null && DateTime.UtcNow < _tokenExpiresAt) return _cachedToken;

            var form = new FormUrlEncodedContent(new Dictionary<string, string>
            {
                ["grant_type"] = "client_credentials",
                ["client_id"] = O.ClientId,
                ["client_secret"] = O.ClientSecret
            });
            using var res = await http.PostAsync($"{O.BaseUrl}/realms/{O.Realm}/protocol/openid-connect/token", form, ct);
            res.EnsureSuccessStatusCode();
            var json = await res.Content.ReadFromJsonAsync<JsonElement>(ct);

            _cachedToken = json.GetProperty("access_token").GetString()!;
            var expiresIn = json.GetProperty("expires_in").GetInt32();
            _tokenExpiresAt = DateTime.UtcNow.AddSeconds(Math.Max(expiresIn - 30, 10));
            return _cachedToken;
        }
        finally
        {
            TokenLock.Release();
        }
    }
}

import { HttpClient } from '@angular/common/http';
import { Injectable, computed, inject, signal } from '@angular/core';
import { Router } from '@angular/router';
import { tap } from 'rxjs';
import { settings } from './app-settings';
import { Portal, Session } from './models';

const STORAGE_KEY = 'helpbot.session';

@Injectable({ providedIn: 'root' })
export class AuthService {
  private http = inject(HttpClient);
  private router = inject(Router);

  private readonly _session = signal<Session | null>(this.restore());
  readonly session = this._session.asReadonly();

  readonly isLoggedIn = computed(() => {
    const s = this._session();
    return !!s && new Date(s.expiresAt).getTime() > Date.now();
  });
  readonly isEmployee = computed(() => this.isLoggedIn() && this._session()!.role === settings.employeeRole);
  readonly isApprover = computed(() => this.isLoggedIn() && this._session()!.role === settings.approverRole);

  get token(): string | null {
    return this.isLoggedIn() ? this._session()!.accessToken : null;
  }

  login(username: string, password: string, portal: Portal) {
    return this.http.post<Session>(`${settings.apiUrl}/auth/login`, { username, password, portal })
      .pipe(tap(session => this.store(session)));
  }

  logout(): void {
    const portal = this._session()?.portal ?? 'employee';
    const done = () => { this.clear(); this.router.navigateByUrl(this.loginUrl(portal)); };
    if (!this.token) { done(); return; }
    this.http.post(`${settings.apiUrl}/auth/logout`, {}).subscribe({ next: done, error: done });
  }

  handleUnauthorized(): void {
    const portal = this._session()?.portal ?? 'employee';
    this.clear();
    this.router.navigateByUrl(this.loginUrl(portal));
  }

  homeUrl(): string { return this.isApprover() ? '/it/approvals' : '/chat'; }
  startUrl(): string { return this.isLoggedIn() ? this.homeUrl() : '/login'; }
  loginUrl(portal: Portal): string { return portal === 'it' ? '/it/login' : '/login'; }

  private store(session: Session): void {
    sessionStorage.setItem(STORAGE_KEY, JSON.stringify(session));
    this._session.set(session);
  }

  private clear(): void {
    sessionStorage.removeItem(STORAGE_KEY);
    this._session.set(null);
  }

  private restore(): Session | null {
    try {
      const raw = sessionStorage.getItem(STORAGE_KEY);
      return raw ? JSON.parse(raw) as Session : null;
    } catch {
      return null;
    }
  }
}

import { ApprovalRequest, AuditLog, ChatMessage, ChatReply, DirectoryUser, ServerInfo } from './models';

const API = settings.apiUrl;

@Injectable({ providedIn: 'root' })
export class ApiService {
  private http = inject(HttpClient);

  chat(message: string) { return this.http.post<ChatReply>(`${API}/chat`, { message }); }
  history() { return this.http.get<ChatMessage[]>(`${API}/chat/history`); }
  myRequests() { return this.http.get<ApprovalRequest[]>(`${API}/chat/my-requests`); }

  approvals() { return this.http.get<ApprovalRequest[]>(`${API}/admin/approvals`); }
  decide(id: string, approve: boolean, note?: string) {
    return this.http.post<ApprovalRequest>(`${API}/admin/approvals/${id}/decision`, { approve, note });
  }
  directoryUsers() { return this.http.get<DirectoryUser[]>(`${API}/admin/directory-users`); }
  unlockDirectoryUser(username: string, reason: string) {
    return this.http.post<{ message: string }>(`${API}/admin/directory-users/${username}/unlock`, { reason });
  }
  resetDirectoryPassword(username: string, reason: string) {
    return this.http.post<{ message: string }>(`${API}/admin/directory-users/${username}/reset-password`, { reason });
  }
  servers() { return this.http.get<ServerInfo[]>(`${API}/admin/servers`); }
  audit() { return this.http.get<AuditLog[]>(`${API}/admin/audit`); }
}

      </label>

      @if (error()) { <p class="error">{{ error() }}</p> }

      <button class="primary full" type="submit" [disabled]="loading()">
        {{ loading() ? 'Signing in…' : 'Sign in' }}
      </button>

      <div class="login-footer">
        @if (portal === 'it') {
          <a routerLink="/login">← Employee sign in</a>
        } @else {
          <a routerLink="/it/login">IT staff? Sign in here →</a>
        }
      </div>
    </form>
  </div>

export class LoginComponent {
  private auth = inject(AuthService);
  private router = inject(Router);

  portal: Portal = inject(ActivatedRoute).snapshot.data['portal'] ?? 'employee';
  username = '';
  password = '';
  loading = signal(false);
  error = signal('');

  submit() {
    if (!this.username.trim() || !this.password) {
      this.error.set('Enter your username and password.');
      return;
    }
    this.loading.set(true);
    this.error.set('');

    this.auth.login(this.username.trim(), this.password, this.portal).subscribe({
      next: () => this.router.navigateByUrl(this.auth.homeUrl()),
      error: e => {
        this.loading.set(false);
        this.password = '';
        this.error.set(e.status === 429
          ? 'Too many attempts. Please wait a minute.'
          : (e.error?.error ?? 'Sign-in failed. Is the API running?'));
      }
    });
  }
}

export class AdminComponent implements OnInit, OnDestroy {
  private api = inject(ApiService);

  approvals = signal<ApprovalRequest[]>([]);
  directoryUsers = signal<DirectoryUser[]>([]);
  servers = signal<ServerInfo[]>([]);
  audit = signal<AuditLog[]>([]);
  busy = signal<string | null>(null);
  error = signal('');
  info = signal('');
  notes: Record<string, string> = {};

  pending = computed(() => this.approvals().filter(a => a.status === 'Pending'));
  decided = computed(() => this.approvals().filter(a => a.status !== 'Pending').slice(0, 15));
  private timer?: ReturnType<typeof setInterval>;

  ngOnInit() {
    this.refresh();
    this.timer = setInterval(() => this.refresh(), 5000);
  }

  ngOnDestroy() { clearInterval(this.timer); }

  refresh() {
    const fail = (e: any) => this.error.set(e.error?.error ?? 'Failed to load data.');
    this.api.approvals().subscribe({ next: a => this.approvals.set(a), error: fail });
    this.api.directoryUsers().subscribe({ next: u => this.directoryUsers.set(u), error: fail });
    this.api.servers().subscribe({ next: s => this.servers.set(s), error: fail });
    this.api.audit().subscribe({ next: a => this.audit.set(a), error: fail });
  }

  decide(r: ApprovalRequest, approve: boolean) {
    this.run(r.id, this.api.decide(r.id, approve, this.notes[r.id]), approve ? 'Request approved.' : 'Request rejected.');
  }

  unlock(u: DirectoryUser) {
    const reason = prompt(`Reason for unlocking ${u.username}'s directory account:`);
    if (!reason) return;
    this.run(u.username, this.api.unlockDirectoryUser(u.username, reason));
  }

  resetPassword(u: DirectoryUser) {
    if (!confirm(`Reset ${u.username}'s Keycloak password? A temporary password will be emailed to them.`)) return;
    const reason = prompt('Reason for the reset (saved in the audit log):');
    if (!reason) return;
    this.run(u.username, this.api.resetDirectoryPassword(u.username, reason));
  }

  args(json: string) {
    try { return Object.entries(JSON.parse(json)).map(([k, v]) => `${k}: ${v}`).join(' · ') || 'no arguments'; }
    catch { return json; }
  }

  private run(key: string, call: any, fallbackMessage = 'Done.') {
    this.busy.set(key);
    this.error.set('');
    this.info.set('');
    call.subscribe({
      next: (res: any) => { this.busy.set(null); this.info.set(res?.message ?? fallbackMessage); this.refresh(); },
      error: (e: any) => { this.busy.set(null); this.error.set(e.error?.error ?? 'Action failed.'); }
    });
  }
}

Before running the API, make sure helpdesk.db is deleted, so the new AppUsers table gets created and seeded. The user-secrets (Llm:ApiKey, Keycloak:ClientSecret, Jwt:SigningKey) stay the same.

Demo flow (the two identities working independently)
Lock the directory account: at http://localhost:8080/realms/helpdesk/account, enter a wrong password for alice 3 times. Alice’s Keycloak account is now locked.
She can still use the helpdesk: log in to http://localhost:4200/login as alice / AlicePortal@1. It works, because the portal is independent of Keycloak.
Self-service unlock: in chat, ask “Is my Keycloak account locked? If so, unlock it.” The agent unlocks it (medium risk, audited).
Password reset with approval: ask “I forgot my Keycloak password, please reset it.” The request becomes PENDING.
IT approves: in a new tab, log in at http://localhost:4200/it/login as itadmin / ItAdmin@123 and approve.
Temporary password: open Mailpit (http://localhost:8025) to find the temporary password.
Keycloak forces a new password: log in at the Keycloak account console with the temporary password. Keycloak shows its own “Update password” screen; set a new password and you’re in.
Portal unchanged: alice’s portal password is still AlicePortal@1.
Direct IT actions: in the IT dashboard’s “Directory accounts” panel, itadmin can also Unlock or Reset password for bob directly, with a mandatory reason that is written to the audit log.
